Integration

Automate Test Transactions

Quickly mock payment requests in your frontend and backend test suites.

// Sample JavaScript test transaction using CC-GEN dummy card
const testPaymentPayload = {
    cardNumber: "5425233489204412",
    expMonth: "12",
    expYear: "2028",
    cvv: "849",
    amount: 4999, // $49.99 USD in cents
    currency: "USD",
    cardholder: "ALEXANDER D. MORGAN"
};

async function processSandboxPayment(payload) {
    const response = await fetch('/api/sandbox/checkout', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
    });
    return await response.json();
}
